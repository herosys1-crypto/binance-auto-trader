DB 스키마:

-- alembic 0025
CREATE TABLE market_flow_records (
    id BIGSERIAL PRIMARY KEY,
    symbol VARCHAR(30), date DATE,
    rank_type VARCHAR(10),  -- pump / dump
    rank INTEGER, change_pct DECIMAL(10, 2),
    features JSONB NOT NULL,  -- flow + indicators + pivots!
    UNIQUE(symbol, date)
);

CREATE TABLE flow_pattern_matches (
    id BIGSERIAL PRIMARY KEY,
    matched_symbol VARCHAR(30), matched_time TIMESTAMPTZ,
    similar_records BIGINT[], similarity_score DECIMAL(4, 3),
    alerted BOOLEAN DEFAULT FALSE
);

DB 스키마:

-- alembic 0026
CREATE TABLE timezone_pivots (
    id BIGSERIAL PRIMARY KEY, symbol VARCHAR(30),
    pivot_time_kst TIMESTAMP,       -- 한국 시간!
    pivot_hour_kst INTEGER,          -- 0~23
    pivot_dow_kst INTEGER,           -- 0=월~6=일
    pivot_type VARCHAR(10),          -- pump / dump
    price_change_pct DECIMAL(10, 2),
    volume_ratio DECIMAL(10, 2)
);

CREATE TABLE timezone_stats (
    id BIGSERIAL PRIMARY KEY, symbol VARCHAR(30),
    hour_kst INTEGER, pump_pct DECIMAL(5, 2), dump_pct DECIMAL(5, 2),
    total_pivots INTEGER, period_days INTEGER,
    last_calculated TIMESTAMPTZ,
    UNIQUE(symbol, hour_kst)
);
